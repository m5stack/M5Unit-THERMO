81inline uint16_t celsius_to_raw(const float f)
82{
83 // Clamp in float before converting, so that NaN or a huge value does not overflow the integer conversion
84 const float v{std::fmax(std::fmin((f + 64) * 128, 65535.f), 0.f)};
85 return static_cast<uint16_t>(std::round(v));
86}
92inline float raw_to_celsius(const uint16_t u16)
93{
94 return u16 / 128.0f - 64;
95}
