196 inline IIR iir() const

202 inline Output output() const
203 {

208 inline FIR fir() const

214 inline Gain gain() const

217 return static_cast<Gain>(g > 6 ? 6 : g);
218 }
221 inline IRSensor irSensor() const
222 {
223 return static_cast<IRSensor>((value >> 6) & 0x01);
224 }
227 inline bool positiveKs() const
228 {
229 return !(value & (1U << 7));

233 inline bool positiveKt2() const
234 {
235 return !(value & (1U << 14));

240 inline void iir(const IIR iir)
241 {
242 value = static_cast<uint16_t>((value & ~0x07) | m5::stl::to_underlying(iir));
243 }
246 inline void output(const Output o)
247 {
248 value = static_cast<uint16_t>((value & ~(0x03 << 4)) | (m5::stl::to_underlying(o) << 4));
249 }
252 inline void fir(const FIR fir)
253 {
254 value = static_cast<uint16_t>((value & ~(0x07 << 8)) | (m5::stl::to_underlying(fir) << 8));
255 }
258 inline void gain(const Gain gain)
259 {

261 }
264 inline void irSensor(const IRSensor irs)
265 {
266 value = static_cast<uint16_t>((value & ~(1U << 6)) | (static_cast<unsigned>(m5::stl::to_underlying(irs)) << 6));
267 }
270 inline void positiveKs(const bool pos)
271 {

273 }
276 inline void positiveKt2(const bool pos)
